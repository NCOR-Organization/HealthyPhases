1) "How does self-transcendence differ from gerotranscendence?"

SELECT ?term ?label ?prefLabel ?definition ?parent ?parentLabel
WHERE {
  VALUES ?wantedLabel { "self-transcendence" "gerotranscendence" }

  ?term a <http://www.w3.org/2002/07/owl#Class> ;
        rdfs:label ?label .
  OPTIONAL { ?term skos:prefLabel ?prefLabel . }
  OPTIONAL { ?term skos:definition ?skosDef . }
  OPTIONAL { ?term obo:IAO_0000115 ?iaoDef . }
  BIND(COALESCE(?skosDef, ?iaoDef) AS ?definition)

  OPTIONAL {
    ?term rdfs:subClassOf ?parent .
    OPTIONAL { ?parent rdfs:label ?parentLabel . }
  }

  FILTER(LCASE(STR(?prefLabel)) = ?wantedLabel || LCASE(STR(?label)) = ?wantedLabel)
}
ORDER BY ?wantedLabel ?parentLabel

2) "What PHASES concepts are in the solitude branch?"

SELECT ?term ?label ?prefLabel ?definition
WHERE {
  ?term a <http://www.w3.org/2002/07/owl#Class> ;
        rdfs:label ?label .
  OPTIONAL { ?term skos:prefLabel ?prefLabel . }
  OPTIONAL { ?term skos:definition ?skosDef . }
  OPTIONAL { ?term obo:IAO_0000115 ?iaoDef . }
  BIND(COALESCE(?skosDef, ?iaoDef) AS ?definition)

  FILTER(STRSTARTS(STR(?term), "http://purl.obolibrary.org/obo/PHASES_"))
  FILTER(
    CONTAINS(LCASE(STR(?label)), "solitude") ||
    CONTAINS(LCASE(STR(?prefLabel)), "solitude")
  )
}
ORDER BY ?label

3) "Which PHASES concepts are most mentioned across papers?"

SELECT ?term ?termLabel (COUNT(*) AS ?mentionCount) (COUNT(DISTINCT ?path) AS ?paperCount)
WHERE {
  ?occ a phases-doc:LexicalOccurrence ;
       phases-doc:lexical_occurrence_of ?term ;
       phases-doc:lexical_occurrence_in_chunk ?chunk .
  ?chunk phases-doc:chunk_of ?paper .
  ?paper phases-doc:path ?path .
  OPTIONAL { ?term rdfs:label ?termLabel . }

  FILTER(STRSTARTS(STR(?term), "http://purl.obolibrary.org/obo/PHASES_"))
}
GROUP BY ?term ?termLabel
ORDER BY DESC(?paperCount) DESC(?mentionCount)
LIMIT 50

8) "Which extracted axioms repeat across multiple papers?"

SELECT ?axiomHash (COUNT(*) AS ?occurrences) (COUNT(DISTINCT ?path) AS ?paperCount)
       (SAMPLE(?axiomText) AS ?exampleAxiom)
WHERE {
  ?ax a phases-ax:ExtractedAxiom ;
      phases-ax:axiom_hash ?axiomHash ;
      phases-ax:axiom_text ?axiomText ;
      phases-ax:axiom_from_chunk ?chunk .
  ?chunk phases-doc:chunk_of ?paper .
  ?paper phases-doc:path ?path .
}
GROUP BY ?axiomHash
HAVING (COUNT(DISTINCT ?path) > 1)
ORDER BY DESC(?paperCount) DESC(?occurrences)
LIMIT 100
